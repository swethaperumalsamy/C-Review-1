#include <stdio.h>


typedef struct
{
    int id;
    char name[30];
    float marks;
} Student;

int main()
{
   
    Student s1;

    s1.id = 101;

    strcpy(s1.name, "Swetha");

    s1.marks = 95.5;

    printf("ID: %d\n", s1.id);
    printf("Name: %s\n", s1.name);
    printf("Marks: %.1f\n", s1.marks);

    return 0;
}
