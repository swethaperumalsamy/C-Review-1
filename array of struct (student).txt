#include <stdio.h>


struct Student
{
    int rollNo;
    char name[30];
    float marks;
};

int main()
{
    struct Student s[5];

    int i, topperIndex = 0;


    for(i = 0; i < 5; i++)
    {
        printf("\nEnter Roll Number: ");
        scanf("%d", &s[i].rollNo);

        printf("Enter Name: ");
        scanf("%s", s[i].name);

        printf("Enter Marks: ");
        scanf("%f", &s[i].marks);
    }

   
    for(i = 1; i < 5; i++)
    {
        if(s[i].marks > s[topperIndex].marks)
        {
            topperIndex = i;
        }
    }

    
    printf("\nTopper Details:\n");

    printf("Roll Number: %d\n", s[topperIndex].rollNo);

    printf("Name: %s\n", s[topperIndex].name);

    printf("Marks: %.2f\n", s[topperIndex].marks);

    return 0;
}
