#include <stdio.h>

// Function to check Sudoku validity
int checkSudoku(int a[9][9])
{
    int i, j, k;

    // Check rows
    for(i = 0; i < 9; i++)
    {
        int seen[10] = {0};

        for(j = 0; j < 9; j++)
        {
            int num = a[i][j];

            if(num < 1 || num > 9 || seen[num])
            {
                return 0;
            }

            seen[num] = 1;
        }
    }

    // Check columns
    for(j = 0; j < 9; j++)
    {
        int seen[10] = {0};

        for(i = 0; i < 9; i++)
        {
            int num = a[i][j];

            if(num < 1 || num > 9 || seen[num])
            {
                return 0;
            }

            seen[num] = 1;
        }
    }

    return 1;
}

int main()
{
    int sudoku[9][9] =
    {
        {5,3,4,6,7,8,9,1,2},
        {6,7,2,1,9,5,3,4,8},
        {1,9,8,3,4,2,5,6,7},
        {8,5,9,7,6,1,4,2,3},
        {4,2,6,8,5,3,7,9,1},
        {7,1,3,9,2,4,8,5,6},
        {9,6,1,5,3,7,2,8,4},
        {2,8,7,4,1,9,6,3,5},
        {3,4,5,2,8,6,1,7,9}
    };

    if(checkSudoku(sudoku))
    {
        printf("Valid Sudoku");
    }
    else
    {
        printf("Invalid Sudoku");
    }

    return 0;
}
