#include <stdio.h>


struct Employee
{
    int id;
    char name[30];
    float salary;
};

int main()
{
    struct Employee emp[10];

    int i, highestIndex = 0;

    float totalSalary = 0;

    
    for(i = 0; i < 10; i++)
    {
        printf("\nEnter Employee ID: ");
        scanf("%d", &emp[i].id);

        printf("Enter Employee Name: ");
        scanf("%s", emp[i].name);

        printf("Enter Salary: ");
        scanf("%f", &emp[i].salary);
    }

        for(i = 0; i < 10; i++)
    {
        totalSalary = totalSalary + emp[i].salary;

        if(emp[i].salary > emp[highestIndex].salary)
        {
            highestIndex = i;
        }
    }


    printf("\nTotal Salary Payout = %.2f\n", totalSalary);

    
    printf("\nEmployee with Highest Salary:\n");

    printf("ID: %d\n", emp[highestIndex].id);

    printf("Name: %s\n", emp[highestIndex].name);

    printf("Salary: %.2f\n", emp[highestIndex].salary);

    return 0;
}
