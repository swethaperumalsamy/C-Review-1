#include <stdio.h>

int main()
{
    FILE *fp1, *fp2;
    int a[100], n = 0, i, j, temp;

    fp1 = fopen("input.txt", "r");

    while(fscanf(fp1, "%d", &a[n]) != EOF)
    {
        n++;
    }

    fclose(fp1);

        for(i = 0; i < n - 1; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(a[i] > a[j])
            {
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }
    }

    fp2 = fopen("output.txt", "w");

    for(i = 0; i < n; i++)
    {
        fprintf(fp2, "%d ", a[i]);
    }

    fclose(fp2);

    return 0;
}
